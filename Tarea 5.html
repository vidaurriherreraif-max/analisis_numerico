Ejercicio 2

Elige 3 indicadores económicos del INEGI (del archivo TIE_Inegi_long.xlsx), realiza la gráfica de la serie temporal y de su derivada. Brinda un análisis de la gráfica y de la evolución del indicador a lo largo del tiempo.

df_inegi = pd.read_excel("TIE_Inegi_long.xlsx")

col_fecha = df_inegi.columns[0]
indicadores = [c for c in df_inegi.columns[1:] if df_inegi[c].dtype != object][:3]
colores_ind = ['steelblue', 'darkred', 'darkorange']

for indicador, color in zip(indicadores, colores_ind):
    serie = df_inegi[[col_fecha, indicador]].dropna()
    fechas = pd.to_datetime(serie[col_fecha])
    valores = serie[indicador].values
    t = np.arange(len(valores), dtype=float)
    derivada = np.gradient(valores, t)

    fig, axes = plt.subplots(2, 1, figsize=(10, 7), sharex=True)
    axes[0].plot(fechas, valores, color=color, lw=1.8)
    axes[0].set_ylabel('Valor del indicador')
    axes[0].set_title(f'{indicador} — Serie temporal')
    axes[0].grid(True)
    axes[1].plot(fechas, derivada, color='black', lw=1.5)
    axes[1].axhline(0, color='gray', lw=0.8, ls='--')
    axes[1].set_ylabel('Tasa de cambio (derivada numérica)')
    axes[1].set_title(f'{indicador} — Derivada numérica')
    axes[1].grid(True)
    plt.tight_layout()
    plt.show()

    print(f"\n=== Análisis: {indicador} ===")
    print(f"Periodo: {fechas.min().date()} — {fechas.max().date()}")
    print(f"Valor mínimo: {valores.min():.4f}  |  Valor máximo: {valores.max():.4f}")
    print(f"Tendencia general: {'creciente' if valores[-1] > valores[0] else 'decreciente'}")
    print(f"Derivada promedio: {derivada.mean():.4f}")
    print(f"Mayor aceleración positiva en índice {derivada.argmax()} "
          f"(fecha aprox.: {fechas.iloc[derivada.argmax()].date()})")
    print(f"Mayor caída en índice {derivada.argmin()} "
          f"(fecha aprox.: {fechas.iloc[derivada.argmin()].date()})")

    h = (b - a) / n
    xi = np.array([a + (i + 0.5)*h for i in range(n)])

def regla_trapecio(f, a, b, n):
    h = (b - a) / n
    xi = np.linspace(a, b, n + 1)
    return h * (f(xi[0])/2 + np.sum(f(xi[1:-1])) + f(xi[-1])/2)
